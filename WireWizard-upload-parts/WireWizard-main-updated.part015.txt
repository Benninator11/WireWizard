      function buildPortListFromForm(device) {
        const selected = [];
        const reusedPortIds = new Set();
        const rows = [...ui.portConfig.children];
        rows.forEach((row) => {
          const typeSelect = row.querySelector('[data-role="type"]');
          const sideSelect = row.querySelector('[data-role="side"]');
          const speedSelect = row.querySelector('[data-role="speed"]');
          const amountInput = row.querySelector('[data-role="amount"]');
          if (!typeSelect || !sideSelect || !speedSelect || !amountInput) return;

          const portType = typeSelect.value;
          const speed = speedSelect.value || 'Standard';
          const amount = Math.max(1, Number(amountInput.value || 1));
          const label = formatPortLabel(portType, speed);

          for (let i = 0; i < amount; i++) {
            const matchedPort = device
              ? device.ports.find((p) => p.type === portType && p.speed === speed && p.side === sideSelect.value && !reusedPortIds.has(p.id))
              : null;
            if (matchedPort) reusedPortIds.add(matchedPort.id);
            const portId = matchedPort ? matchedPort.id : uid('port');
            selected.push({
              id: portId,
              type: portType,
              label,
              side: sideSelect.value,
              speed,
              amount,
              cableId: matchedPort ? matchedPort.cableId : null,
            });
          }
        });

        return selected.length ? selected : [{ id: uid('port'), type: 'Schuko', label: 'Schuko', side: 'left', speed: 'Standard', amount: 1, cableId: null }];
      }

      function buildSocketPorts(device, outletCount, deviceType = 'Steckdosenleiste') {
        const previousPorts = device ? device.ports || [] : [];
        const existingInput = previousPorts.find((port) => port.powerRole === 'input');
        const existingOutlets = previousPorts.filter((port) => port.powerRole === 'outlet');
        const rackPower = ['Rack-PDU', 'USV'].includes(deviceType);
        const inputType = rackPower ? 'IEC C20' : 'Schuko';
        const outletType = rackPower ? 'IEC C13' : 'Schuko';
        const inputPort = {
          ...(existingInput || {}),
          id: existingInput ? existingInput.id : uid('port'),
          type: inputType,
          label: rackPower ? 'Einspeisung IEC C20' : 'Stromanschluss',
          side: 'left',
          speed: 'Standard',
          powerRole: 'input',
          cableId: existingInput ? existingInput.cableId : null,
        };
        const outlets = Array.from({ length: outletCount }, (_, index) => {
          const outletNumber = index + 1;
          const previous = existingOutlets.find((port) => port.outletNumber === outletNumber);
          return {
            ...(previous || {}),
            id: previous ? previous.id : uid('port'),
            type: outletType,
            label: rackPower ? `IEC C13 Ausgang ${outletNumber}` : `Steckdose ${outletNumber}`,
            side: 'right',
            speed: 'Standard',
            powerRole: 'outlet',
            outletNumber,
            amount: 1,
            cableId: previous ? previous.cableId : null,
          };
        });
        return [inputPort, ...outlets];
      }

      function buildDevicePorts(device, deviceType) {
        if (deviceType === 'Kabelorganisator') return [];
        if (deviceType === 'Steckdose') return buildSocketPorts(device, 1, deviceType).slice(1);
        if (powerDistributionTypes.has(deviceType) && deviceType !== 'Steckdose') {
          const outletCount = clamp(Math.floor(Number(ui.outletCountInput.value) || 1), 1, 20);
          ui.outletCountInput.value = outletCount;
          return buildSocketPorts(device, outletCount, deviceType);
        }
        return buildPortListFromForm(device);
      }

      function hasRemovedConnectedPorts(existing, nextPorts) {
        if (!existing) return false;
        const nextPortIds = new Set(nextPorts.map((port) => port.id));
        return existing.ports.some((port) => port.cableId && !nextPortIds.has(port.id));
      }

      function saveDeviceFromForm(event) {
        event.preventDefault();
        if (!ui.deviceName.value.trim()) {
          showToast(t('deviceNameRequired'));
          return;
        }

        const deviceInput = {
          id: state.openDeviceId || uid('device'),
          name: ui.deviceName.value.trim(),
          description: ui.deviceDescription.value.trim(),
          storage: Number(ui.deviceStorageValue.value || 0),
          storageUnit: ui.deviceStorageUnit.value,
          cpuModel: ui.cpuModel.value.trim(),
          ramGb: Number(ui.ramGb.value || 0),
          ramType: ui.ramType.value,
          cpuCores: Number(ui.cpuCores.value || 0),
          cpuClockGHz: Number(ui.cpuClockGHz.value || 0),
          gpuModel: ui.gpuModel.value.trim(),
          gpuCount: clamp(Math.floor(Number(ui.gpuCount.value) || 1), 1, 16),
          cpuThreads: plannerNumberFromSpec(getPlannerCpuHardwareSpec(ui.cpuModel.value)?.threads),
          cpuBaseClockGHz: plannerClockGHz(getPlannerCpuHardwareSpec(ui.cpuModel.value)?.baseClock),
          cpuBoostClockGHz: plannerClockGHz(getPlannerCpuHardwareSpec(ui.cpuModel.value)?.boostClock),
          hardwareSpecs: getPlannerStoredHardwareSpecs(ui.cpuModel.value, ui.gpuModel.value),
          powerWatt: Number(ui.devicePowerWatt.value || 0),
          type: ui.deviceType.value,
          w: 190,
          h: powerDistributionTypes.has(ui.deviceType.value) && ui.deviceType.value !== 'Steckdose'
            ? Math.max(150, 110 + clamp(Math.floor(Number(ui.outletCountInput.value) || 1), 1, 20) * 25)
            : (ui.deviceType.value === 'Steckdose' ? 160 : 135),
          ports: []
        };

        const existing = state.devices.get(deviceInput.id);
        deviceInput.ports = buildDevicePorts(existing, deviceInput.type);
        const portsBySide = deviceInput.ports.reduce((counts, port) => {
          const side = port.side || 'right';
          counts[side] = (counts[side] || 0) + 1;
          return counts;
        }, {});
        const verticalPortCount = Math.max(portsBySide.left || 0, portsBySide.right || 0);
        const horizontalPortCount = Math.max(portsBySide.top || 0, portsBySide.bottom || 0);
        deviceInput.h = Math.max(deviceInput.h, verticalPortCount * 26 + 26);
        deviceInput.w = Math.max(deviceInput.w, horizontalPortCount * 26 + 26);
        if (deviceInput.type !== 'Kabelorganisator' && !hasPowerInput(deviceInput.ports, deviceInput.powerWatt)) {
          showToast(t('missingPowerPort'));
          return;
        }
        if (existing && hasRemovedConnectedPorts(existing, deviceInput.ports)) {
          showToast(t('removeOutletCableFirst'));
          return;
        }
        if (existing) {
          deviceInput.x = existing.x;
          deviceInput.y = existing.y;
          state.devices.set(deviceInput.id, { ...existing, ...deviceInput });
        } else {
          const pos = randomPosition(deviceInput.w, deviceInput.h);
          deviceInput.x = pos.x;
          deviceInput.y = pos.y;
          state.enteringDeviceId = deviceInput.id;
          state.devices.set(deviceInput.id, deviceInput);
        }

        saveState();
        render();
        state.enteringDeviceId = null;
        closeDeviceModal();
        showToast(t('saved'));
      }

      function deleteSelectedDevice() {
        const selectedId = state.selectedDeviceId;
        if (!selectedId) {
          showToast(t('noDeviceSelected'));
          return;
        }
        const device = state.devices.get(selectedId);
        if (!device) return;
        const relatedCableIds = [...state.cables.keys()].filter((cableId) => {
          const cable = state.cables.get(cableId);
          return cable && (cable.fromDeviceId === device.id || cable.toDeviceId === device.id);
        });
        relatedCableIds.forEach((cableId) => state.cables.delete(cableId));
        state.devices.delete(selectedId);
        if (state.hoveredDeviceId === selectedId) state.hoveredDeviceId = null;
        state.selectedDeviceId = null;
        state.selectedCableId = null;
        saveState();
        render();
      }

      function deleteSelectedCable() {
        const cableId = state.selectedCableId;
        if (!cableId) return;
        state.cables.delete(cableId);
        state.devices.forEach((device) => {
          device.ports.forEach((port) => {
            if (port.cableId === cableId) port.cableId = null;
          });
        });
        state.selectedCableId = null;
        saveState();
        render();
      }

            </div>
            <span class="small-button">✎</span>

          item.addEventListener('click', () => {
            state.selectedDeviceId = device.id;
            openDeviceModal(device.id);
            renderDeviceList();
          });
          ui.deviceList.appendChild(item);
        });
      }

        ui.deviceLayer.innerHTML = '';
        state.devices.forEach((device) => {
          const node = document.createElement('div');

          if (state.enteringDeviceId === device.id) node.classList.add('device-entering');
          node.style.left = `${device.x}px`;
          node.style.top = `${device.y}px`;
          node.style.width = `${device.w}px`;
          node.style.height = `${device.h}px`;
          node.dataset.deviceId = device.id;

          node.addEventListener('mouseenter', () => {
            state.hoveredDeviceId = device.id;
          });
          node.addEventListener('mouseleave', () => {
            if (state.hoveredDeviceId === device.id) state.hoveredDeviceId = null;
          });

          copy.className = 'device-copy';

          description.className = 'device-description';
          description.textContent = device.description || '—';
          const power = document.createElement('div');
          power.className = 'device-power';
          power.textContent = `${Number(device.powerWatt || 0)} W`;
