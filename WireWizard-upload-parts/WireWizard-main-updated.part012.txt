      function saveState() {
        const payload = {
          lang: state.lang,
          theme: state.theme,
          zoom: state.zoom,
          devices: Array.from(state.devices.values()),
          cables: Array.from(state.cables.values()),
          selectedCableType: state.selectedCableType,
          nextCounter: state.nextCounter,
        };
        localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
      }

      function loadState() {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (!raw) return;
        try {
          const parsed = JSON.parse(raw);
          if (!parsed || !Array.isArray(parsed.devices)) return;
          state.lang = parsed.lang || 'de';
          state.theme = parsed.theme === 'dark' ? 'dark' : 'light';
          state.zoom = Number.isFinite(Number(parsed.zoom)) ? clamp(Number(parsed.zoom), MIN_ZOOM, MAX_ZOOM) : 1;
          state.selectedCableType = parsed.selectedCableType || 'LAN';
          state.nextCounter = Number(parsed.nextCounter || 1);
          state.devices = new Map((parsed.devices || []).map((device) => [device.id, device]));
          state.cables = new Map((parsed.cables || []).map((cable) => [cable.id, cable]));
        } catch (err) {
          console.error('Stand konnte nicht geladen werden:', err);
        }
      }

      function ensurePortCableIds() {
        state.devices.forEach((device) => {
          device.ports.forEach((port) => {
            const related = [...state.cables.values()].find((cable) => {
              return (cable.fromDeviceId === device.id && cable.fromPortId === port.id) || (cable.toDeviceId === device.id && cable.toPortId === port.id);
            });
            port.cableId = related ? related.id : null;
          });
        });
      }

      function clamp(value, min, max) {
        return Math.min(Math.max(value, min), max);
      }

      function isPowerPort(portType) {
        return powerPortTypes.has((portType || '').trim().toLowerCase());
