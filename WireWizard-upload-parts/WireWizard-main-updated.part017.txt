      // Dragging mit Maus oder Touch.
      function startDragging(event, deviceId, deviceNode) {
        const device = getDeviceById(deviceId);
        if (!device || event.pointerId === undefined) return;

        const pointerId = event.pointerId;
        const startX = event.clientX;
        const startY = event.clientY;
        const initialX = device.x;
        const initialY = device.y;
        let moved = false;

        const move = (moveEvent) => {
          if (moveEvent.pointerId !== pointerId) return;
          const deltaX = moveEvent.clientX - startX;
          const deltaY = moveEvent.clientY - startY;
          if (Math.abs(deltaX) + Math.abs(deltaY) > 4) moved = true;
          const nextX = clamp(initialX + deltaX / state.zoom, 10, CANVAS_WIDTH - device.w - 10);
          const nextY = clamp(initialY + deltaY / state.zoom, 10, CANVAS_HEIGHT - device.h - 10);
          device.x = nextX;
          device.y = nextY;

          // Element beim Ziehen nicht neu bauen, sonst reißt das Touch-Capture ab.
          // Neu gezeichnet werden nur die Kabel.
          if (deviceNode?.isConnected) {
            deviceNode.style.left = `${device.x}px`;
            deviceNode.style.top = `${device.y}px`;
          }
          renderCables();
          if (moveEvent.cancelable) moveEvent.preventDefault();
        };

        const stop = (upEvent) => {
          if (upEvent && upEvent.pointerId !== pointerId) return;
          document.removeEventListener('pointermove', move);
          document.removeEventListener('pointerup', stop);
          document.removeEventListener('pointercancel', stop);
          try {
            if (deviceNode?.hasPointerCapture(pointerId)) deviceNode.releasePointerCapture(pointerId);
          } catch (err) { /* Pointer war schon freigegeben */ }
          saveState();

          if (moved) {
            const suppressClick = (clickEvent) => {
              clickEvent.preventDefault();
              clickEvent.stopPropagation();
              ui.deviceLayer.removeEventListener('click', suppressClick, true);
            };
            ui.deviceLayer.addEventListener('click', suppressClick, true);
            window.setTimeout(() => ui.deviceLayer.removeEventListener('click', suppressClick, true), 350);
          }
        };

        document.addEventListener('pointermove', move, { passive: false });
        document.addEventListener('pointerup', stop);
        document.addEventListener('pointercancel', stop);
      }

      function exportJson() {
        const payload = JSON.stringify({
          devices: Array.from(state.devices.values()),
          cables: Array.from(state.cables.values()),
          lang: state.lang,
          theme: state.theme,
          zoom: state.zoom,
          selectedCableType: state.selectedCableType,
          maxPowerWatt: state.maxPowerWatt,
        }, null, 2);
        const blob = new Blob([payload], { type: 'application/json' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = 'serverplaner.json';
        link.click();
        URL.revokeObjectURL(url);
      }

      function importJsonFile(file) {
        if (!file) return;
        const reader = new FileReader();
        reader.onload = (event) => {
          try {
            const parsed = JSON.parse(String(event.target.result));
            if (!parsed || !Array.isArray(parsed.devices)) throw new Error('invalid');
            state.devices = new Map((parsed.devices || []).map((device) => [device.id, device]));
            state.cables = new Map((parsed.cables || []).map((cable) => [cable.id, cable]));
            state.lang = parsed.lang || state.lang;
            if (parsed.theme === 'dark' || parsed.theme === 'light') state.theme = parsed.theme;
            if (Number.isFinite(Number(parsed.zoom))) state.zoom = clamp(Number(parsed.zoom), MIN_ZOOM, MAX_ZOOM);
            state.selectedCableType = parsed.selectedCableType || state.selectedCableType;
            state.maxPowerWatt = parsed.maxPowerWatt || MAX_POWER_WATT;
            state.selectedDeviceId = null;
            state.selectedCableId = null;
            state.selectedPort = null;
            applyZoom(false);
            saveState();
            render();
            showToast(t('imported'));
          } catch (err) {
            console.error('JSON-Import fehlgeschlagen:', err);
            showToast(t('invalidJson'));
          }
        };
        reader.readAsText(file);
      }

      function applyZoom(preserveCenter = true) {
        const shell = ui.canvas.parentElement;
        const shellRect = shell.getBoundingClientRect();
        const contentRect = ui.canvasContent.getBoundingClientRect();
        const contentLeft = contentRect.left - shellRect.left - shell.clientLeft + shell.scrollLeft;
        const contentTop = contentRect.top - shellRect.top - shell.clientTop + shell.scrollTop;
        const previousZoom = Number(ui.canvas.dataset.zoom || 1);
        const focusX = (shell.scrollLeft + shell.clientWidth / 2 - contentLeft) / previousZoom;
        const focusY = (shell.scrollTop + shell.clientHeight / 2 - contentTop) / previousZoom;

        ui.canvas.style.width = `${CANVAS_WIDTH * state.zoom}px`;
        ui.canvas.style.height = `${CANVAS_HEIGHT * state.zoom}px`;
        ui.canvasContent.style.transform = `scale(${state.zoom})`;
        ui.canvas.style.backgroundSize = `${22 * state.zoom}px ${22 * state.zoom}px`;
        ui.canvas.dataset.zoom = String(state.zoom);

        if (preserveCenter && previousZoom !== state.zoom) {
          shell.scrollLeft = contentLeft + focusX * state.zoom - shell.clientWidth / 2;
          shell.scrollTop = contentTop + focusY * state.zoom - shell.clientHeight / 2;
        }
        ui.zoomLevel.textContent = `${Math.round(state.zoom * 100)}%`;
        ui.zoomOutBtn.disabled = state.zoom <= MIN_ZOOM;
