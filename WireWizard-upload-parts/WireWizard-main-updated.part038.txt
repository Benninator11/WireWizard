          const description = document.createElement('div');
          description.className = 'device-description';
          description.textContent = device.description || '—';
          const power = document.createElement('div');
          power.className = 'device-power';
          power.textContent = `${Number(device.powerWatt || 0)} W`;
          copy.appendChild(description);
          copy.appendChild(power);
          body.appendChild(model);
          body.appendChild(copy);
          const outletPorts = device.ports.filter((port) => port.powerRole === 'outlet');
          const outletMeterRows = outletPorts.map((port) => `
            <div class="outlet-meter-row">
              <span>${getPortDisplayLabel(port.label)}</span>
              <strong>${calculateOutletLoad(device, port)} W</strong>
            </div>
          `).join('');
          if (outletMeterRows) {
            const meterList = document.createElement('div');
            meterList.className = 'outlet-meter-list';
            meterList.innerHTML = outletMeterRows;
            body.appendChild(meterList);
          }

          node.appendChild(header);
          node.appendChild(body);

          const portsBySide = { left: [], right: [], top: [], bottom: [] };
          device.ports.forEach((port) => {
            const side = port.side || 'right';
            if (portsBySide[side]) portsBySide[side].push(port);
          });

          Object.entries(portsBySide).forEach(([side, list]) => {
            list.forEach((port, index) => {
              const btn = document.createElement('button');
              btn.type = 'button';
              btn.className = 'port' + (port.cableId ? ' used' : '');
              btn.dataset.deviceId = device.id;
              btn.dataset.portId = port.id;
              btn.dataset.side = side;
              btn.textContent = port.outletNumber ? `S${port.outletNumber}` : getPortShortLabel(port.type, port.speed || port.label);
              if (port.powerRole === 'outlet') btn.title = `${getPortDisplayLabel(port.label)}: ${calculateOutletLoad(device, port)} W`;

              let x = 0;
              let y = 0;
              const total = Math.max(list.length, 1);
              const ratio = (index + 1) / (total + 1);
              if (side === 'left') {
                x = 0;
                y = (device.h * ratio);
              } else if (side === 'right') {
                x = device.w;
                y = (device.h * ratio);
              } else if (side === 'top') {
                x = (device.w * ratio);
                y = 0;
              } else {
                x = (device.w * ratio);
                y = device.h;
              }
              btn.style.left = `${x}px`;
              btn.style.top = `${y}px`;

              btn.addEventListener('click', (event) => {
                event.stopPropagation();
                handlePortClick(device.id, port.id);
              });

              node.appendChild(btn);
            });
          });

          node.addEventListener('pointerdown', (event) => {
            if (event.isPrimary === false || (event.pointerType === 'mouse' && event.button !== 0)) return;
            if (event.target.closest('.port, button, input, select, textarea, a')) return;
            state.selectedDeviceId = device.id;
            state.selectedCableId = null;
            node.classList.add('selected');
            try { node.setPointerCapture(event.pointerId); } catch (err) { /* Pointer schon weg */ }
            startDragging(event, device.id, node);
          });

          node.addEventListener('click', () => {
            state.selectedDeviceId = device.id;
            openDeviceModal(device.id);
            renderDeviceList();
          });

          ui.deviceLayer.appendChild(node);
        });
      }

      function pathBetweenPorts(from, to) {
        const dx = Math.max(90, Math.abs(to.x - from.x) * 0.75);
        const cx1 = from.x + dx;
        const cy1 = from.y;
        const cx2 = to.x - dx;
        const cy2 = to.y;
        return `M ${from.x} ${from.y} C ${cx1} ${cy1}, ${cx2} ${cy2}, ${to.x} ${to.y}`;
      }

      function getPortWorldPosition(device, port) {
        const portsBySide = { left: [], right: [], top: [], bottom: [] };
        device.ports.forEach((candidate) => {
          const side = candidate.side || 'right';
          if (portsBySide[side]) portsBySide[side].push(candidate);
        });
        const list = portsBySide[port.side] || [];
        const index = list.findIndex((candidate) => candidate.id === port.id);
        const total = Math.max(list.length, 1);
        const ratio = (index + 1) / (total + 1);

        let x = 0;
        let y = 0;

        if (port.side === 'left') {
          x = device.x;
          y = device.y + device.h * ratio;
        } else if (port.side === 'right') {
          x = device.x + device.w;
          y = device.y + device.h * ratio;
        } else if (port.side === 'top') {
          x = device.x + device.w * ratio;
          y = device.y;
        } else {
          x = device.x + device.w * ratio;
          y = device.y + device.h;
        }

        return { x, y };
      }

      function renderCables() {
        ui.cableLayer.innerHTML = '';
        state.cables.forEach((cable) => {
          const fromDevice = getDeviceById(cable.fromDeviceId);
          const toDevice = getDeviceById(cable.toDeviceId);
          const fromPort = fromDevice ? getPortById(cable.fromDeviceId, cable.fromPortId) : null;
          const toPort = toDevice ? getPortById(cable.toDeviceId, cable.toPortId) : null;
          if (!fromDevice || !toDevice || !fromPort || !toPort) return;

          const start = getPortWorldPosition(fromDevice, fromPort);
          const end = getPortWorldPosition(toDevice, toPort);
          const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
          path.setAttribute('d', pathBetweenPorts(start, end));
          path.setAttribute('stroke', cable.color || getCableColor(cable.type));
          path.setAttribute('stroke-width', state.selectedCableId === cable.id ? '6' : '4');
          path.setAttribute('fill', 'none');
          path.setAttribute('class', 'cable-path' + (state.selectedCableId === cable.id ? ' selected' : ''));
          path.setAttribute('aria-label', getCableTypeLabel(cable.type));
          const title = document.createElementNS('http://www.w3.org/2000/svg', 'title');
          title.textContent = getCableTypeLabel(cable.type);
          path.appendChild(title);
          path.style.pointerEvents = 'auto';
          path.addEventListener('pointermove', (event) => {
            const tooltip = ui.cableTooltip;
            tooltip.textContent = getCableTypeLabel(cable.type);
            tooltip.classList.add('visible');
            tooltip.setAttribute('aria-hidden', 'false');
            tooltip.style.left = `${Math.max(8, Math.min(event.clientX + 14, window.innerWidth - tooltip.offsetWidth - 8))}px`;
            tooltip.style.top = `${Math.max(8, Math.min(event.clientY + 14, window.innerHeight - tooltip.offsetHeight - 8))}px`;
          });
          path.addEventListener('pointerleave', () => {
            ui.cableTooltip.classList.remove('visible');
            ui.cableTooltip.setAttribute('aria-hidden', 'true');
          });
          path.addEventListener('contextmenu', (event) => {
            event.preventDefault();
            state.selectedCableId = cable.id;
            ui.cableColorPicker.value = cable.color || getCableColor(cable.type);
            if (typeof ui.cableColorPicker.showPicker === 'function') ui.cableColorPicker.showPicker();
            else ui.cableColorPicker.click();
          });
          path.addEventListener('click', (event) => {
            event.stopPropagation();
            state.selectedCableId = cable.id;
            state.selectedDeviceId = null;
            render();
          });
          ui.cableLayer.appendChild(path);
        });
      }

      function handlePortClick(deviceId, portId) {
        const device = getDeviceById(deviceId);
        const port = getPortById(deviceId, portId);
        if (!device || !port) return;

        if (!state.selectedPort) {
          if (port.cableId) {
            showToast(t('occupiedPort'));
            return;
          }
          const compatibleCableTypes = getCompatibleCableTypes(port.type)
            .filter((type) => isCableCompatible(port, type, device));
          if (!compatibleCableTypes.includes(state.selectedCableType)) {
            state.selectedCableType = compatibleCableTypes[0] || 'Sonstiges';
            ui.cableTypeSelect.value = state.selectedCableType;
            saveState();
          }
          state.selectedPort = { deviceId, portId };
          const portNode = document.querySelector(`.port[data-device-id="${deviceId}"][data-port-id="${portId}"]`);
          if (portNode) portNode.classList.add('active');
          return;
        }

        const currentA = state.selectedPort;
        const deviceA = getDeviceById(currentA.deviceId);
        const portA = getPortById(currentA.deviceId, currentA.portId);
        const portB = port;

        if (portA && portB && portA.powerRole === 'outlet' && portB.powerRole === 'outlet') {
          showToast(t('incompatibleCable'));
          state.selectedPort = null;
          render();
          return;
        }

        if (currentA.deviceId === deviceId && currentA.portId === portId) {
          showToast(t('selfConnectionBlocked'));
          state.selectedPort = null;
          render();
          return;
        }

        const commonCableTypes = portA && portB
          ? getCompatibleCableTypes(portA.type).filter((type) => {
            return getCompatibleCableTypes(portB.type).includes(type)
              && isCableCompatible(portA, type, deviceA)
              && isCableCompatible(portB, type, device);
          })
          : [];
        const cableType = commonCableTypes.includes(state.selectedCableType)
          ? state.selectedCableType
          : commonCableTypes[0];
        if (!cableType) {
          showToast(t('incompatibleCable'));
          state.selectedPort = null;
          render();
          return;
        }

        if (portA && portB && (portA.cableId || portB.cableId)) {
          showToast(t('occupiedPort'));
          state.selectedPort = null;
          render();
          return;
        }

        const cableId = uid('cable');
        const cable = {
          id: cableId,
          type: cableType,
          color: getCableColor(cableType),
          fromDeviceId: currentA.deviceId,
          fromPortId: currentA.portId,
          toDeviceId: deviceId,
          toPortId: portId,
        };

        state.cables.set(cableId, cable);
        portA.cableId = cableId;
        portB.cableId = cableId;
        state.selectedCableType = cableType;
        state.selectedPort = null;
        saveState();
        render();
      }

          'Strom': 'Power',

          Fiber: 'Fiber optic',
          DAC: 'DAC (Direct Attach Copper)',
          Cat6: 'Cat6',
          'Sonstiges': 'Other'
        };
        return labels[type] || type;
      }

      function getPortDisplayLabel(label) {
        if (state.lang !== 'en' || !label) return label || '';
        if (label === 'Stromanschluss') return 'Power input';
        if (label === 'Einspeisung IEC C20') return 'IEC C20 power input';
        let match = label.match(/^Steckdose (\d+)$/);
        if (match) return `Power outlet ${match[1]}`;
        match = label.match(/^IEC C13 Ausgang (\d+)$/);
        if (match) return `IEC C13 output ${match[1]}`;
        return label;
      }
