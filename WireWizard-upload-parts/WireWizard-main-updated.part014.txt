      function getPlannerCpuHardwareSpec(model) {
        const name = String(model || '').trim();
        if (!name) return null;
        return plannerCpuSpecsByExactName[name.toLocaleLowerCase()] || plannerCpuSpecsByNormalizedName[normalizePlannerHardwareName(name)] || null;
      }
      
      function getPlannerGpuHardwareSpec(model) {
        const name = String(model || '').trim();
        if (!name) return null;
        return plannerGpuSpecsByExactName[name.toLocaleLowerCase()] || plannerGpuSpecsByNormalizedName[normalizePlannerHardwareName(name)] || null;
      }
      
      function plannerNumberFromSpec(value) {
        const text = String(value ?? '').trim();
        if (!text || /^(?:—|-|n\/a|nicht ermittelt|nicht verifiziert|not verified|unknown)$/i.test(text)) return 0;
        const values = [...text.matchAll(/\d+(?:[.,]\d+)?/g)].map(match => Number(match[0].replace(',', '.'))).filter(number => Number.isFinite(number) && number > 0);
        return values.length ? Math.max(...values) : 0;
      }
      
      function plannerCoreCount(spec) {
        if (!spec) return 0;
        const text = String(spec.cores || '').trim();
        if (!text || /nicht ermittelt|nicht verifiziert|not verified|unknown|^—$|^-$/.test(text.toLocaleLowerCase())) return 0;
        const pair = text.match(/^\s*(\d+)\s*\+\s*(\d+)/);
        if (pair && !text.includes('(')) return Number(pair[1]) + Number(pair[2]);
        const count = text.match(/^\s*(\d+)/);
        return count ? Number(count[1]) : 0;
      }
      
      function plannerClockGHz(value) {
        return plannerNumberFromSpec(value);
      }
      
      function plannerWattFromSpec(value) {
        const text = String(value || '').replace(/(\d),(?=\d{3}\b)/g, '$1');
        if (!text || /nicht verifiziert|not verified|nicht ermittelt|unknown|unbekannt/i.test(text)) return 0;
        const numbers = [];
        const regex = /(\d+(?:\.\d+)?)\s*(?:[-–—]\s*(\d+(?:\.\d+)?))?\s*(?:W(?:att)?s?\b)/gi;
        let match;
        while ((match = regex.exec(text)) !== null) {
          numbers.push(Number(match[1]));
          if (match[2]) numbers.push(Number(match[2]));
        }
        return numbers.filter(number => Number.isFinite(number) && number > 0).reduce((max, number) => Math.max(max, number), 0);
      }
      
      function plannerSelectedCpuClockGHz(spec) {
        if (!spec) return 0;
        return plannerClockGHz(spec.boostClock) || plannerClockGHz(spec.baseClock);
      }

          ui.cpuClockGHz.value = clock ? String(clock) : (onlyFillEmpty ? ui.cpuClockGHz.value : '');
        }
      }

          || /\bIntel Core Ultra\b.*\b\d{3,5}(?:U|H|V|P)\b/i.test(name)
          || /\bRyzen AI\b/i.test(name);

          const closeList = () => {

            const availableNames = typeof names === 'function' ? names() : names;
            const query = input.value.trim().toLocaleLowerCase();

            });
            list.replaceChildren(options);
            list.classList.remove('hidden');
            input.setAttribute('aria-expanded', 'true');
          };

          input.addEventListener('input', renderOptions);

            if (event.key === 'Escape') closeList();

              event.preventDefault();

            }

              event.preventDefault();
              list.firstElementChild.click();
            }
          });
          list.addEventListener('keydown', (event) => {
            if (event.key === 'Escape') {
              closeList();
              input.focus();
            }
          });
          document.addEventListener('click', (event) => {
            if (!wrapper.contains(event.target)) closeList();
          });
        };
        bindSearchList('deviceCpuModel', 'cpuNameOptions', () => getCompatibleCpuNames(ui.deviceType.value));
        bindSearchList('deviceGpuModel', 'gpuNameOptions', () => getCompatibleGpuNames(ui.deviceType.value));
      }

      function getPortSpeedOptions(portType) {
        const base = portType.toLowerCase();

        if (base.includes('lan')) {
          return ['100 Mbps', '1 GbE', '2.5 GbE', '10 GbE'];
        }
        if (base.includes('usb')) return ['1.1', '2.0', '3.0', '3.1', '3.2', 'USB-C'];
        if (base.includes('thunderbolt')) {
          return ['3', '4', '5'];
        }
        if (base.includes('hdmi')) return ['HDMI 1.4', 'HDMI 2.0', 'HDMI 2.1'];
        if (base.includes('displayport')) return ['1.2', '1.4', '2.0', '2.1'];
        if (base.includes('vga')) return ['VGA'];
        if (base.includes('dvi')) return ['DVI-D', 'DVI-I'];
        if (base.includes('klinke') || base.includes('cinch') || base.includes('xlr')) return ['3,5 mm', '6,3 mm', 'RCA', 'XLR'];
        if (base.includes('sfp')) return ['SFP', 'SFP+'];
        if (base.includes('sd')) return ['SD', 'SDHC', 'SDXC'];
        if (base.includes('lightning')) return ['Lightning'];
        return ['Standard'];
      }

      function formatPortLabel(portType, speed) {
        if (!speed || speed === 'Standard') return portType;
        return `${portType} ${speed}`;
      }

      function getPortShortLabel(portType, speed) {
        const type = (portType || '').trim();
        const speedText = ((speed || '').trim() || type).trim();
        const lowerType = type.toLowerCase();

        if (lowerType === 'usb-c') return 'UC';
        if (lowerType.includes('usb') || lowerType.startsWith('usb')) {
          const match = (speedText.match(/\d+(?:\.\d+)?/) || [''])[0];
          const main = match.includes('.') ? match.split('.')[0] : match;
          return match ? `U${main}` : 'U';
        }
        if (lowerType.includes('lan')) return 'L';
        if (lowerType.includes('thunderbolt')) return 'T';
        if (lowerType.includes('hdmi')) return 'H';
        if (lowerType.includes('displayport')) return 'DP';
        if (lowerType.includes('vga')) return 'V';
        if (lowerType.includes('dvi')) return 'D';
        if (lowerType.includes('sfp')) return 'S';
        if (lowerType.includes('lightning')) return 'Li';
        if (lowerType.includes('sd')) return 'SD';
        return type.slice(0, 2).toUpperCase();
      }

      const cableColorPalette = [
        '#e63946', '#0077b6', '#2a9d8f', '#f4a261', '#7b2cbf', '#d63384',

      function hueToHex(hue) {
        const chroma = 0.7 * (1 - Math.abs(2 * 0.46 - 1));
        const segment = hue / 60;
        const secondary = chroma * (1 - Math.abs(segment % 2 - 1));
        const offset = 0.46 - chroma / 2;
        let rgb;
        if (segment < 1) rgb = [chroma, secondary, 0];
        else if (segment < 2) rgb = [secondary, chroma, 0];
        else if (segment < 3) rgb = [0, chroma, secondary];
        else if (segment < 4) rgb = [0, secondary, chroma];
        else if (segment < 5) rgb = [secondary, 0, chroma];
        else rgb = [chroma, 0, secondary];
        return `#${rgb.map((value) => Math.round((value + offset) * 255).toString(16).padStart(2, '0')).join('')}`;
      }

      function getUniqueCableColor() {
        const usedColors = new Set([...state.cables.values()].map((cable) => {
          return (cable.color || getCableColor(cable.type)).toLowerCase();
        }));
        const paletteColor = cableColorPalette.find((color) => !usedColors.has(color));
        if (paletteColor) return paletteColor;

        for (let index = 0; index < 360; index++) {
          const color = hueToHex((index * 137.508) % 360);
          if (!usedColors.has(color)) return color;
        }
        return '#000000';
      }

      function randomPosition(w = 190, h = 135) {
        const margin = 40;
        const shell = ui.canvas.parentElement;
        const contentRect = ui.canvasContent.getBoundingClientRect();
        const shellRect = shell.getBoundingClientRect();
        const scaleX = contentRect.width / ui.canvasContent.offsetWidth;
        const scaleY = contentRect.height / ui.canvasContent.offsetHeight;
        const visibleLeft = clamp(Math.max(contentRect.left, shellRect.left + shell.clientLeft, 0), contentRect.left, contentRect.right);
        const visibleRight = clamp(Math.min(contentRect.right, shellRect.left + shell.clientLeft + shell.clientWidth, window.innerWidth), contentRect.left, contentRect.right);
        const visibleTop = clamp(Math.max(contentRect.top, shellRect.top + shell.clientTop, 0), contentRect.top, contentRect.bottom);
        const visibleBottom = clamp(Math.min(contentRect.bottom, shellRect.top + shell.clientTop + shell.clientHeight, window.innerHeight), contentRect.top, contentRect.bottom);
        const visibleMinX = clamp((visibleLeft - contentRect.left) / scaleX, 0, ui.canvasContent.offsetWidth);
        const visibleMaxX = clamp((visibleRight - contentRect.left) / scaleX, 0, ui.canvasContent.offsetWidth);
        const visibleMinY = clamp((visibleTop - contentRect.top) / scaleY, 0, ui.canvasContent.offsetHeight);
        const visibleMaxY = clamp((visibleBottom - contentRect.top) / scaleY, 0, ui.canvasContent.offsetHeight);
        const canvasMaxX = Math.max(0, ui.canvasContent.offsetWidth - w);
        const canvasMaxY = Math.max(0, ui.canvasContent.offsetHeight - h);
        let minX = Math.max(margin, visibleMinX);
        let maxX = Math.min(canvasMaxX - margin, visibleMaxX - w);
        let minY = Math.max(margin, visibleMinY);
        let maxY = Math.min(canvasMaxY - margin, visibleMaxY - h);

        if (maxX < minX) minX = maxX = clamp((visibleMinX + visibleMaxX - w) / 2, 0, canvasMaxX);
        if (maxY < minY) minY = maxY = clamp((visibleMinY + visibleMaxY - h) / 2, 0, canvasMaxY);

        let x = minX + Math.random() * Math.max(0, maxX - minX);
        let y = minY + Math.random() * Math.max(0, maxY - minY);
        let attempts = 0;
        while (attempts < 300) {
          const collision = [...state.devices.values()].some((device) => {
            return !(x + w < device.x - 30 || x > device.x + device.w + 30 || y + h < device.y - 30 || y > device.y + device.h + 30);
          });
          if (!collision) return { x: clamp(x, minX, maxX), y: clamp(y, minY, maxY) };
          x = minX + Math.random() * Math.max(0, maxX - minX);
          y = minY + Math.random() * Math.max(0, maxY - minY);
          attempts++;
        }
        return { x: minX, y: minY };
      }
