  function isCoreNetworkDevice(device) {

  function hasNetworkPath(startDevice) {
    if (isCoreNetworkDevice(startDevice)) return true;
    const targets = new Set([...state.devices.values()].filter(isCoreNetworkDevice).map((device) => device.id));
    if (!targets.size) return false;
    const queue = [startDevice.id], seen = new Set(queue);
    while (queue.length) {
      const id = queue.shift();
      const current = state.devices.get(id);
      if (!current) continue;
      for (const port of current.ports || []) {
        if (!isNetworkPort(port) || isPowerPort(port)) continue;
        const endpoint = getConnectedEndpoint(current, port);
        if (!endpoint || !endpoint.device || !isNetworkPort(endpoint.port) || isPowerPort(endpoint.port)) continue;
        const nextId = endpoint.device.id;
        if (targets.has(nextId)) return true;
        if (!seen.has(nextId)) { seen.add(nextId); queue.push(nextId); }
      }
    }
    return false;
  }

    if (!distributionDevices.length) return originalCalculateTotalPower();

      if (!device || visited.has(device.id)) return 0;
      if (/Rack-PDU|USV|Steckdose|PDU/i.test(String(device.type || '') + ' ' + String(device.name || ''))) {

          if (endpoint) sum += load(endpoint.device, visited);

        return sum;

      return Number(device.powerWatt) || 0;
