  function serverRackPlacement(device, capacity = serverRackCapacity()) {
    const start = Number(device.rackStartU);
    const height = serverRackHeight(device);
    if (!Number.isInteger(start) || start < 1 || start + height - 1 > capacity) return null;
    return { start, top: start + height - 1, height };
  }
  function serverRackCanPlace(deviceId, start, capacity = serverRackCapacity()) {
    const device = state.devices.get(deviceId);
    if (!device) return false;
    const height = serverRackHeight(device);
    if (start < 1 || start + height - 1 > capacity) return false;
    for (const other of state.devices.values()) {
      if (other.id === deviceId) continue;
      const otherPlacement = serverRackPlacement(other, capacity);
      if (!otherPlacement) continue;
      if (start <= otherPlacement.top && start + height - 1 >= otherPlacement.start) return false;
    }
    return true;
  }
  function serverRackPlace(deviceId, topUnit) {
    const device = state.devices.get(deviceId);
    if (!device) return;
    const height = serverRackHeight(device);
    const start = Number(topUnit) - height + 1;
    if (start < 1) {
      showToast(state.lang === 'en' ? 'Not enough rack units above the bottom.' : 'Das Gerät passt an dieser Position nicht vollständig in den Schrank.');
      return;
    }
    if (!serverRackCanPlace(deviceId, start)) {
      showToast(state.lang === 'en' ? 'These rack units are already occupied.' : 'Diese Rack-Units sind bereits belegt.');
      return;
    }
    device.rackStartU = start;
    selectedRackDeviceId = null;
    saveState();
    render();
    renderServerRack();
    showToast(state.lang === 'en' ? `${device.name} placed at U${start}–U${topUnit}.` : `${device.name} auf U${start}–U${topUnit} platziert.`);

      block.title = `${device.name} · U${placement.start}–U${placement.top} · ${device.type}`;
      const name = document.createElement('strong');
      name.textContent = device.name || (state.lang === 'en' ? 'Device' : 'Gerät');
      const type = document.createElement('small');
      type.textContent = `${device.type} · ${placement.height}U`;
      block.append(name, type);
      block.addEventListener('dragstart', (event) => {
        event.dataTransfer.setData('text/x-serverwizard-device', device.id);
        event.dataTransfer.setData('text/plain', device.id);
        event.dataTransfer.effectAllowed = 'move';
      });
      block.addEventListener('click', (event) => {
        event.stopPropagation();
        selectedRackDeviceId = device.id;
        renderServerRack();
      });
      rackMounted.appendChild(block);
    });

    rackPool.replaceChildren();
    const devices = [...state.devices.values()].sort((a, b) => String(a.name || '').localeCompare(String(b.name || ''), state.lang === 'en' ? 'en' : 'de'));
    rackEmpty.hidden = devices.length > 0;
    devices.forEach((device) => {
      const placement = serverRackPlacement(device, capacity);
      const isSelected = selectedRackDeviceId === device.id;
      const card = document.createElement('div');
      card.className = 'rack-pool-card' + (isSelected ? ' selected-to-place' : '');
      card.tabIndex = 0;
      card.setAttribute('role', 'button');
      card.draggable = true;
      const main = document.createElement('div');
      main.className = 'rack-pool-card-main';
      const name = document.createElement('strong');
      name.textContent = device.name || (state.lang === 'en' ? 'Device' : 'Gerät');
      const details = document.createElement('small');
      details.textContent = placement
        ? `${device.type} · U${placement.start}–U${placement.top}`
        : `${device.type} · ${serverRackHeight(device)}U · ${state.lang === 'en' ? 'not placed' : 'nicht platziert'}`;
      main.append(name, details);
      card.appendChild(main);
      card.addEventListener('click', () => {
        selectedRackDeviceId = device.id;
        renderServerRack();
      });
      card.addEventListener('keydown', (event) => {
        if (event.key === 'Enter' || event.key === ' ') { event.preventDefault(); selectedRackDeviceId = device.id; renderServerRack(); }
      });
      card.addEventListener('dragstart', (event) => {
        event.dataTransfer.setData('text/x-serverwizard-device', device.id);
        event.dataTransfer.setData('text/plain', device.id);
        event.dataTransfer.effectAllowed = 'move';
      });
      if (placement) {
        const remove = document.createElement('button');
        remove.type = 'button';
        remove.textContent = '×';
        remove.title = state.lang === 'en' ? 'Remove from rack' : 'Aus dem Rack nehmen';
        remove.setAttribute('aria-label', remove.title);
        remove.addEventListener('click', (event) => { event.stopPropagation(); serverRackRemove(device.id); });
        card.appendChild(remove);
      }
      rackPool.appendChild(card);
    });
    const placedCount = visiblePlaced.size;
    const title = rackOverlay.querySelector('.server-rack-stage-title span:first-child');
    title.textContent += ` · ${placedCount} ${state.lang === 'en' ? 'placed' : 'platziert'}`;
  }

    rackOverlay.hidden = false;
    rackButton.setAttribute('aria-pressed', 'true');

  }

    rackOverlay.hidden = true;
    rackButton.setAttribute('aria-pressed', 'false');
    selectedRackDeviceId = null;
  }
