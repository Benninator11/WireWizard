      cleanedSavedDevices = true;
    }
    if (noPower.has(device.type) && Number(device.powerWatt) !== 0) {
      device.powerWatt = 0;
      cleanedSavedDevices = true;

      device.extraBaseHeight = Math.max(135, Number(device.extraBaseHeight) || Number(device.h) || 135);
      device.h = Math.max(device.extraBaseHeight, Math.max(1, Number(device.rackU) || 1) * 35);
      cleanedSavedDevices = true;
    }
  });
  if (cleanedSavedDevices) {
    saveState();
    render();

    let selectedDeviceMode = 'auto';
    let currentLanguage = 'de';

    function detectDeviceMode() {
      const width = window.innerWidth;
      const ua = navigator.userAgent;
      const isPhone = /iPhone|iPod|Android.*Mobile|Windows Phone|IEMobile|Opera Mini|Mobile/i.test(ua);
      const isTablet = /iPad|Tablet|Android(?!.*Mobile)/i.test(ua);
      // iPadOS meldet sich als Mac
      const isIpadAsMac = navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1;
      const coarsePointer = window.matchMedia('(pointer: coarse)').matches;

      if (isPhone || width <= 650) return 'mobile';
      if (isTablet || isIpadAsMac || width <= 900 || (coarsePointer && width <= 1200)) return 'tablet';
      return 'pc';
    }

    function resolvedDeviceMode() {
      return selectedDeviceMode === 'auto' ? detectDeviceMode() : selectedDeviceMode;
    }

    function syncDeviceModeToFrame(frame) {
      if (!frame.contentWindow) return;
      frame.contentWindow.postMessage({ type: 'planner-display-mode', mode: resolvedDeviceMode() }, '*');
    }

    function applyDeviceMode(mode) {
      if (!['auto', 'pc', 'tablet', 'mobile'].includes(mode)) return;
      selectedDeviceMode = mode;
      const resolved = resolvedDeviceMode();
      document.documentElement.dataset.deviceMode = resolved;
      document.body.dataset.deviceMode = resolved;
      deviceModeButtons.forEach((button) => {
        const active = button.dataset.deviceModeChoice === mode;
        button.classList.toggle('active', active);
        button.setAttribute('aria-pressed', String(active));
      });

      const en = currentLanguage === 'en';
      const names = { pc: 'PC', tablet: 'Tablet', mobile: en ? 'Phone' : 'Handy' };
      const prefix = mode === 'auto' ? (en ? 'Auto:' : 'Automatisch:') : (en ? 'Mode:' : 'Modus:');
      deviceModeStatus.textContent = `${prefix} ${names[resolved]}`;

      syncDeviceModeToFrame(serverView);
      syncDeviceModeToFrame(cableView);
    }

    deviceModeButtons.forEach((button) => {
      button.addEventListener('click', () => applyDeviceMode(button.dataset.deviceModeChoice));
    });
    ['resize', 'orientationchange'].forEach((name) => {
      window.addEventListener(name, () => {
        if (selectedDeviceMode === 'auto') applyDeviceMode('auto');
