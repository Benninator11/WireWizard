(function setupCanvasPanning() {
  const viewport = document.querySelector('.canvas-shell');
  const canvas = document.querySelector('.canvas');
  if (!viewport || !canvas || viewport.dataset.leftDragPanReady === 'true') return;
  viewport.dataset.leftDragPanReady = 'true';

  // Die ganze Canvas verschieben, nicht nur die Geräte, sonst laufen Raster
  // und Kabel auseinander. Der Zoom-Transform bleibt unberührt.
  const readTranslate = () => {
    const value = getComputedStyle(canvas).translate || 'none';
    const parts = value.match(/-?\d+(?:\.\d+)?px/g) || [];
    return { x: Number.parseFloat(parts[0]) || 0, y: Number.parseFloat(parts[1]) || 0 };
  };
  const initial = readTranslate();
  let panX = initial.x;
  let panY = initial.y;
  let drag = null;
  let spaceHeld = false;
  let didMove = false;

  const isEditable = (target) => !!target?.closest('input, textarea, select, [contenteditable="true"]');
  const isInteractive = (target) => !!target?.closest(
    '.device, .port, .cable-path, button, input, select, textarea, a, .modal, .help-overlay, .visualizer-mode, [data-no-pan]'
  );

  document.addEventListener('keydown', (event) => {
    if (event.code === 'Space' && !isEditable(event.target)) spaceHeld = true;
  });
  document.addEventListener('keyup', (event) => {
    if (event.code === 'Space') spaceHeld = false;
  });
  window.addEventListener('blur', () => { spaceHeld = false; finishPan(); });

  viewport.addEventListener('pointerdown', (event) => {
    if (event.button !== 0 || event.isPrimary === false) return;
    const target = event.target instanceof Element ? event.target : null;
    if (isInteractive(target) && !spaceHeld) return;

    drag = { pointerId: event.pointerId, startX: event.clientX, startY: event.clientY, x: panX, y: panY };
    didMove = false;
    try { viewport.setPointerCapture(event.pointerId); } catch (err) { /* s.o. */ }
    viewport.style.cursor = 'grabbing';
    viewport.style.userSelect = 'none';
    event.preventDefault();
    // bei Leertaste+Ziehen soll das Gerät darunter stehen bleiben
    event.stopPropagation();
  }, true);

  viewport.addEventListener('pointermove', (event) => {
    if (!drag || event.pointerId !== drag.pointerId) return;
    const dx = event.clientX - drag.startX;
    const dy = event.clientY - drag.startY;
    if (Math.abs(dx) + Math.abs(dy) > 2) didMove = true;
    panX = drag.x + dx;
    panY = drag.y + dy;
    canvas.style.translate = `${panX}px ${panY}px`;
    event.preventDefault();
  }, true);

  function finishPan() {
    if (!drag) return;
    const pointerId = drag.pointerId;
    drag = null;
    viewport.style.cursor = '';
    viewport.style.userSelect = '';
    try { if (viewport.hasPointerCapture(pointerId)) viewport.releasePointerCapture(pointerId); } catch (err) { /* s.o. */ }
    if (didMove) {
      // nach dem Verschieben soll der Klick nichts (ab)wählen
      const suppressClick = (event) => { event.preventDefault(); event.stopPropagation(); viewport.removeEventListener('click', suppressClick, true); };
      viewport.addEventListener('click', suppressClick, true);
      window.setTimeout(() => viewport.removeEventListener('click', suppressClick, true), 250);
    }
  }
  viewport.addEventListener('pointerup', finishPan, true);
  viewport.addEventListener('pointercancel', finishPan, true);
  viewport.addEventListener('lostpointercapture', finishPan, true);
  viewport.style.cursor = 'grab';
})();
