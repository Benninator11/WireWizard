        return '#000000';
      }

      function getCompatibleCableTypes(portType) {
        const type = (portType || '').trim().toLowerCase();
        if (type.includes('usb-pd')) return ['USB-PD', 'Stromkabel'];
        if (type === 'usb-c') return ['USB', 'USB-PD', 'Thunderbolt', 'Stromkabel'];
        if (type.includes('usb')) return ['USB', 'Stromkabel'];
        if (type.includes('lan') || type.includes('ethernet') || type.includes('rj45')) return ['LAN', 'Cat6', 'Fiber', 'DAC', 'PoE'];
        if (type.includes('thunderbolt')) return ['Thunderbolt'];
        if (type.includes('hdmi')) return ['HDMI'];
        if (type.includes('displayport')) return ['DisplayPort'];
        if (type.includes('vga')) return ['VGA'];
        if (type.includes('dvi')) return ['DVI'];
        if (type.includes('klinke') || type.includes('cinch') || type.includes('xlr')) return ['Audio'];
        if (type === 'ps/2') return ['PS/2'];
        if (type === 'serial') return ['Serial'];
        if (type === 'parallel') return ['Parallel'];
        if (type === 'schuko' || type.includes('iec c')) return ['Strom', 'Stromkabel'];
        if (type === 'dc-in') return ['dc-in', 'Stromkabel'];
        if (type.includes('lightning')) return ['Lightning'];
        if (type.includes('sfp')) return ['SFP', 'Fiber', 'DAC'];
        if (type.includes('sd')) return ['SD'];
        if (type === 'cf') return ['CF'];
        if (type === 'cfexpress') return ['CFexpress'];
        return ['Sonstiges'];
      }
      function isCableCompatible(port, cableType, device = null) {
        if (!port || !getCompatibleCableTypes(port.type).includes(cableType)) return false;
        if (cableType === 'PoE') return Number(device && device.powerWatt) <= 100;
        return true;
      }

      function randomPosition(w = 190, h = 135) {
        const margin = 40;
        const shell = ui.canvas.parentElement;
        const contentRect = ui.canvasContent.getBoundingClientRect();
        const shellRect = shell.getBoundingClientRect();
        const scaleX = contentRect.width / ui.canvasContent.offsetWidth;
        const scaleY = contentRect.height / ui.canvasContent.offsetHeight;
        const visibleLeft = clamp(Math.max(contentRect.left, shellRect.left + shell.clientLeft, 0), contentRect.left, contentRect.right);
        const visibleRight = clamp(Math.min(contentRect.right, shellRect.left + shell.clientLeft + shell.clientWidth, window.innerWidth), contentRect.left, contentRect.right);
        const visibleTop = clamp(Math.max(contentRect.top, shellRect.top + shell.clientTop, 0), contentRect.top, contentRect.bottom);
        const visibleBottom = clamp(Math.min(contentRect.bottom, shellRect.top + shell.clientTop + shell.clientHeight, window.innerHeight), contentRect.top, contentRect.bottom);
        const visibleMinX = clamp((visibleLeft - contentRect.left) / scaleX, 0, ui.canvasContent.offsetWidth);
        const visibleMaxX = clamp((visibleRight - contentRect.left) / scaleX, 0, ui.canvasContent.offsetWidth);
        const visibleMinY = clamp((visibleTop - contentRect.top) / scaleY, 0, ui.canvasContent.offsetHeight);
        const visibleMaxY = clamp((visibleBottom - contentRect.top) / scaleY, 0, ui.canvasContent.offsetHeight);
        const canvasMaxX = Math.max(0, ui.canvasContent.offsetWidth - w);
        const canvasMaxY = Math.max(0, ui.canvasContent.offsetHeight - h);
        let minX = Math.max(margin, visibleMinX);
        let maxX = Math.min(canvasMaxX - margin, visibleMaxX - w);
        let minY = Math.max(margin, visibleMinY);
        let maxY = Math.min(canvasMaxY - margin, visibleMaxY - h);

        if (maxX < minX) minX = maxX = clamp((visibleMinX + visibleMaxX - w) / 2, 0, canvasMaxX);
        if (maxY < minY) minY = maxY = clamp((visibleMinY + visibleMaxY - h) / 2, 0, canvasMaxY);

        let x = minX + Math.random() * Math.max(0, maxX - minX);
        let y = minY + Math.random() * Math.max(0, maxY - minY);
        let attempts = 0;
        while (attempts < 300) {
          const collision = [...state.devices.values()].some((device) => {
            return !(x + w < device.x - 30 || x > device.x + device.w + 30 || y + h < device.y - 30 || y > device.y + device.h + 30);
          });
          if (!collision) return { x: clamp(x, minX, maxX), y: clamp(y, minY, maxY) };
          x = minX + Math.random() * Math.max(0, maxX - minX);
          y = minY + Math.random() * Math.max(0, maxY - minY);
          attempts++;
        }
        return { x: minX, y: minY };
      }

      function buildPortListFromForm(device) {
        const selected = [];
        const rows = [...ui.portConfig.children];
        rows.forEach((row) => {
          const typeSelect = row.querySelector('[data-role="type"]');
          const sideSelect = row.querySelector('[data-role="side"]');
          const speedSelect = row.querySelector('[data-role="speed"]');
          const amountInput = row.querySelector('[data-role="amount"]');
          if (!typeSelect || !sideSelect || !speedSelect || !amountInput) return;

          const portType = typeSelect.value;
          const speed = speedSelect.value || 'Standard';
          const amount = Math.max(1, Number(amountInput.value || 1));
          const label = formatPortLabel(portType, speed);

          for (let i = 0; i < amount; i++) {
            const matchedPort = device ? device.ports.find((p) => p.type === portType && p.speed === speed && p.side === sideSelect.value && (!p.cableId || p.cableId === null || true)) : null;
            const portId = matchedPort ? matchedPort.id : uid('port');
            selected.push({
              id: portId,
              type: portType,
              label,
              side: sideSelect.value,
              speed,
              amount,
              cableId: matchedPort ? matchedPort.cableId : null,
            });
          }
        });

        return selected.length ? selected : [{ id: uid('port'), type: 'Schuko', label: 'Schuko', side: 'left', speed: 'Standard', amount: 1, cableId: null }];
      }

      function buildSocketPorts(device, outletCount) {
        const previousPorts = device ? device.ports || [] : [];
        const existingInput = previousPorts.find((port) => port.powerRole === 'input');
        const existingOutlets = previousPorts.filter((port) => port.powerRole === 'outlet');
        const inputPort = {
          ...(existingInput || {}),
          id: existingInput ? existingInput.id : uid('port'),
          type: 'Schuko',
          label: 'Stromanschluss',
          side: 'left',
          speed: 'Standard',
          powerRole: 'input',
          cableId: existingInput ? existingInput.cableId : null,
        };
        const outlets = Array.from({ length: outletCount }, (_, index) => {
          const outletNumber = index + 1;
          const previous = existingOutlets.find((port) => port.outletNumber === outletNumber);
          return {
            ...(previous || {}),
            id: previous ? previous.id : uid('port'),
            type: 'Schuko',
            label: `Steckdose ${outletNumber}`,
            side: 'right',
            speed: 'Standard',
            powerRole: 'outlet',
            outletNumber,
            amount: 1,
            cableId: previous ? previous.cableId : null,
          };
        });
        return [inputPort, ...outlets];
      }

      function buildDevicePorts(device, deviceType) {
        if (deviceType === 'Steckdose') return buildSocketPorts(device, 1).slice(1);
        if (deviceType === 'Steckdosenleiste') {
          const outletCount = clamp(Math.floor(Number(ui.outletCountInput.value) || 1), 1, 20);
          ui.outletCountInput.value = outletCount;
          return buildSocketPorts(device, outletCount);
        }
        return buildPortListFromForm(device);
      }

      function hasRemovedConnectedPorts(existing, nextPorts) {
        if (!existing) return false;
        const nextPortIds = new Set(nextPorts.map((port) => port.id));
        return existing.ports.some((port) => port.cableId && !nextPortIds.has(port.id));
      }

      function saveDeviceFromForm(event) {
        event.preventDefault();
        if (!ui.deviceName.value.trim()) {
          showToast(t('deviceNameRequired'));
          return;
        }

        const deviceInput = {
          id: state.openDeviceId || uid('device'),
          name: ui.deviceName.value.trim(),
          description: ui.deviceDescription.value.trim(),
          storage: ['Steckdose', 'Monitor', 'Fernseher', 'Tastatur', 'Maus', 'USB-Hub', 'Adapter', 'Steckdosenleiste', 'Lampe'].includes(ui.deviceType.value) ? 0 : Number(ui.deviceStorageValue.value || 0),
          storageUnit: ui.deviceStorageUnit.value,
          cpuModel: ui.cpuModel.value.trim(),
          cpuCores: plannerCoreCount(getPlannerCpuHardwareSpec(ui.cpuModel.value)),
          cpuThreads: plannerNumberFromSpec(getPlannerCpuHardwareSpec(ui.cpuModel.value)?.threads),
          cpuClockGHz: plannerSelectedCpuClockGHz(getPlannerCpuHardwareSpec(ui.cpuModel.value)),
          cpuBaseClockGHz: plannerClockGHz(getPlannerCpuHardwareSpec(ui.cpuModel.value)?.baseClock),
          cpuBoostClockGHz: plannerClockGHz(getPlannerCpuHardwareSpec(ui.cpuModel.value)?.boostClock),
          ramGb: Number(ui.ramGb.value || 0),
          ramType: ui.ramType.value,
          gpuModel: ui.gpuModel.value.trim(),
          gpuCount: clamp(Math.floor(Number(ui.gpuCount.value) || 1), 1, 16),
          hardwareSpecs: getPlannerStoredHardwareSpecs(ui.cpuModel.value, ui.gpuModel.value),
          raspberryPiVersion: ui.raspberryPiVersion.value,
          powerWatt: ['Steckdose', 'USB-Hub', 'Adapter', 'Steckdosenleiste'].includes(ui.deviceType.value) ? 0 : Number(ui.devicePowerWatt.value || 0),
          type: ui.deviceType.value,
          w: 190,
          h: ui.deviceType.value === 'Steckdosenleiste'
            ? Math.max(150, 110 + clamp(Math.floor(Number(ui.outletCountInput.value) || 1), 1, 20) * 25)
            : (ui.deviceType.value === 'Steckdose' ? 160 : 135),
          ports: []
        };

        const existing = state.devices.get(deviceInput.id);
        deviceInput.ports = buildDevicePorts(existing, deviceInput.type);
        if (!['Steckdose', 'USB-Hub', 'Adapter', 'Steckdosenleiste'].includes(deviceInput.type) && !hasPowerInput(deviceInput.ports, deviceInput.powerWatt)) {
          showToast(t('missingPowerPort'));
          return;
        }
        if (existing && hasRemovedConnectedPorts(existing, deviceInput.ports)) {
          showToast(t('removeOutletCableFirst'));
          return;
        }
        if (existing) {
          deviceInput.x = existing.x;
          deviceInput.y = existing.y;
          state.devices.set(deviceInput.id, { ...existing, ...deviceInput });
        } else {
          const pos = randomPosition(deviceInput.w, deviceInput.h);
          deviceInput.x = pos.x;
          deviceInput.y = pos.y;
          state.enteringDeviceId = deviceInput.id;
          state.devices.set(deviceInput.id, deviceInput);
        }

        saveState();
        render();
        state.enteringDeviceId = null;
        closeDeviceModal();
        showToast(t('saved'));
      }

      function deleteSelectedDevice() {
        const selectedId = state.selectedDeviceId;
        if (!selectedId) {
          showToast(t('noDeviceSelected'));
          return;
        }
        const device = state.devices.get(selectedId);
        if (!device) return;
        const relatedCableIds = [...state.cables.keys()].filter((cableId) => {
          const cable = state.cables.get(cableId);
          return cable && (cable.fromDeviceId === device.id || cable.toDeviceId === device.id);
        });
        relatedCableIds.forEach((cableId) => state.cables.delete(cableId));
        state.devices.delete(selectedId);
        if (state.hoveredDeviceId === selectedId) state.hoveredDeviceId = null;
        state.selectedDeviceId = null;
        state.selectedCableId = null;
        saveState();
        render();
      }

      function deleteSelectedCable() {
        const cableId = state.selectedCableId;
        if (!cableId) return;
        state.cables.delete(cableId);
        state.devices.forEach((device) => {
          device.ports.forEach((port) => {
            if (port.cableId === cableId) port.cableId = null;
          });
        });
        state.selectedCableId = null;
        saveState();
        render();
      }

          });
          ui.deviceList.appendChild(item);
        });

      // Geräte neu zeichnen. Die Ports müssen danach auch wieder stimmen.
      function renderDevices() {
        ui.deviceLayer.innerHTML = '';
        state.devices.forEach((device) => {
          const node = document.createElement('div');
          node.className = 'device' + (state.selectedDeviceId === device.id ? ' selected' : '');
          node.style.width = Math.max(170, Math.min(330, 150 + (device.ports || []).length * 5)) + 'px';
          if (state.enteringDeviceId === device.id) node.classList.add('device-entering');
          node.style.left = `${device.x}px`;
          node.style.top = `${device.y}px`;
          node.style.width = `${device.w}px`;
          node.style.height = `${device.h}px`;
          node.dataset.deviceId = device.id;

          node.addEventListener('mouseenter', () => {
            state.hoveredDeviceId = device.id;
          });
          node.addEventListener('mouseleave', () => {
            if (state.hoveredDeviceId === device.id) state.hoveredDeviceId = null;
          });

          const header = document.createElement('div');
          header.className = 'device-header';
          const name = document.createElement('div');
          name.className = 'device-name';
          name.textContent = device.name || 'Gerät';
          const chip = document.createElement('span');
          chip.className = 'small-button';
          chip.textContent = getDeviceTypeLabel(device.type);
          header.appendChild(name);
          header.appendChild(chip);

          const body = document.createElement('div');
          body.className = 'device-body';
          const model = document.createElement('div');
          model.className = 'device-model';
          model.innerHTML = getDeviceModelSvg(device.type);
          const copy = document.createElement('div');
          copy.className = 'device-copy';
