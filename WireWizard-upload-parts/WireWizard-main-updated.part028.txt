      function estimateSelectedHardwarePowerWatt(deviceType, cpuModel, gpuModel, gpuCount = 1) {
        const cpu = String(cpuModel || '').trim();
        const gpu = String(gpuModel || '').trim();
        if (!cpu && !gpu) return getDefaultPowerWatt(deviceType);
        const base = hardwareBasePowerWattByType[deviceType] ?? Math.max(15, Math.round(getDefaultPowerWatt(deviceType) * 0.2));
        const count = gpu ? clamp(Math.floor(Number(gpuCount) || 1), 1, 16) : 0;
        const gpuPower = gpu ? estimateGpuPowerWatt(gpu, deviceType) * count : 0;
        return Math.round(base + estimateCpuPowerWatt(cpu, deviceType) + gpuPower);
      }

      function updatePowerEstimateFromHardware() {
        const deviceType = ui.deviceType.value;
        const cpuModel = hardwareSpecTypes.has(deviceType) ? ui.cpuModel.value : '';
        const gpuModel = gpuSpecTypes.has(deviceType) ? ui.gpuModel.value : '';
        const gpuCount = ui.gpuCount ? ui.gpuCount.value : 1;
        ui.devicePowerWatt.value = estimateSelectedHardwarePowerWatt(deviceType, cpuModel, gpuModel, gpuCount);
        ui.devicePowerWatt.dataset.customized = 'false';
      }

      function hasPowerInput(ports, powerWatt) {
        return ports.some((port) => isPowerPort(port.type)) || (
          Number(powerWatt) <= 100 && ports.some((port) => (port.type || '').trim().toLowerCase() === 'lan')
        );
      }

      function calculateTotalPower() {
        let total = 0;
        state.devices.forEach((device) => {
          const watt = Number(device.powerWatt) || 0;
          total += watt;
        });
        return total;
      }

      function getConnectedEndpoint(device, port) {
        const cable = port.cableId ? state.cables.get(port.cableId) : null;
        if (!cable) return null;
        if (cable.fromDeviceId === device.id && cable.fromPortId === port.id) {
          const otherDevice = getDeviceById(cable.toDeviceId);
          return otherDevice ? { device: otherDevice, port: getPortById(otherDevice.id, cable.toPortId) } : null;
        }
        if (cable.toDeviceId === device.id && cable.toPortId === port.id) {
          const otherDevice = getDeviceById(cable.fromDeviceId);
          return otherDevice ? { device: otherDevice, port: getPortById(otherDevice.id, cable.fromPortId) } : null;
        }
        return null;
      }

      function calculateDeviceLoad(device, visited = new Set()) {
        if (!device) return 0;
        if (device.type !== 'Steckdose' && device.type !== 'Steckdosenleiste') return Number(device.powerWatt) || 0;
        if (visited.has(device.id)) return 0;
        visited.add(device.id);
        const total = device.ports
          .filter((port) => port.powerRole === 'outlet')
          .reduce((sum, port) => sum + calculateOutletLoad(device, port, visited), 0);
        visited.delete(device.id);
        return total;
      }

      function calculateOutletLoad(device, port, visited = new Set()) {
        const endpoint = getConnectedEndpoint(device, port);
        return endpoint ? calculateDeviceLoad(endpoint.device, visited) : 0;
      }

      function renderPowerBar() {
        const total = calculateTotalPower();
        const percent = clamp((total / state.maxPowerWatt) * 100, 0, 100);
        const warning = total > state.maxPowerWatt;
        ui.powerFill.style.width = `${percent}%`;
        ui.powerLabel.textContent = `${t('power')}: ${total} / ${state.maxPowerWatt} W`;
        ui.powerWrap.classList.toggle('warning', warning);
        if (warning) {
          ui.powerLabel.title = t('warnOverLimit');
        } else {
          ui.powerLabel.title = 'OK';
        }
      }

      function getPortSpeedOptions(portType) {
        const base = portType.toLowerCase();
        if (base.includes('lan')) return ['100 Mbps', '1 GbE', '2.5 GbE', '10 GbE'];
        if (base.includes('usb')) return ['1.1', '2.0', '3.0', '3.1', '3.2', 'USB-C'];
        if (base.includes('thunderbolt')) return ['3', '4', '5'];
        if (base.includes('hdmi')) return ['HDMI 1.4', 'HDMI 2.0', 'HDMI 2.1'];
        if (base.includes('displayport')) return ['1.2', '1.4', '2.0', '2.1'];
        if (base.includes('vga')) return ['VGA'];
        if (base.includes('dvi')) return ['DVI-D', 'DVI-I'];
        if (base.includes('klinke') || base.includes('cinch') || base.includes('xlr')) return ['3,5 mm', '6,3 mm', 'RCA', 'XLR'];
        if (base.includes('sfp')) return ['SFP', 'SFP+'];
        if (base.includes('sd')) return ['SD', 'SDHC', 'SDXC'];
        if (base.includes('lightning')) return ['Lightning'];
        return ['Standard'];
      }

      function formatPortLabel(portType, speed) {
        if (!speed || speed === 'Standard') return portType;
        return `${portType} ${speed}`;
      }

      function getPortShortLabel(portType, speed) {
        const type = (portType || '').trim();
        const speedText = ((speed || '').trim() || type).trim();
        const lowerType = type.toLowerCase();

        if (lowerType === 'usb-c') return 'UC';
        if (lowerType.includes('usb') || lowerType.startsWith('usb')) {
          const match = (speedText.match(/\d+(?:\.\d+)?/) || [''])[0];
          const main = match.includes('.') ? match.split('.')[0] : match;
          return match ? `U${main}` : 'U';
        }
        if (lowerType.includes('lan')) return 'L';
        if (lowerType.includes('thunderbolt')) return 'T';
        if (lowerType.includes('hdmi')) return 'H';
        if (lowerType.includes('displayport')) return 'DP';
        if (lowerType.includes('vga')) return 'V';
        if (lowerType.includes('dvi')) return 'D';
        if (lowerType.includes('sfp')) return 'S';
        if (lowerType.includes('lightning')) return 'Li';
        if (lowerType.includes('sd')) return 'SD';
        return type.slice(0, 2).toUpperCase();
      }

      function createPortRow(port = null) {
        const row = document.createElement('div');
        row.className = 'port-check';

        const typeSelect = document.createElement('select');
        typeSelect.dataset.role = 'type';
        portCatalog.forEach((portType) => {
          const option = document.createElement('option');
          option.value = portType;
          option.textContent = getPortTypeLabel(portType);
          typeSelect.appendChild(option);
        });
        typeSelect.value = port ? port.type : 'LAN';

        const sideSelect = document.createElement('select');
        sideSelect.dataset.role = 'side';
        sideSelect.innerHTML = ['left', 'right', 'top', 'bottom'].map((side) => `
          <option value="${side}">${getPortSideLabel(side)}</option>
        `).join('');
        sideSelect.value = port ? (port.side || 'right') : 'right';

        const speedSelect = document.createElement('select');
        speedSelect.dataset.role = 'speed';
        const basePortType = (port ? port.type : typeSelect.value);
        const speedOptions = getPortSpeedOptions(basePortType);
        speedOptions.forEach((speed) => {
          const option = document.createElement('option');
          option.value = speed;
          option.textContent = getPortSpeedLabel(speed);
          speedSelect.appendChild(option);
        });
        speedSelect.value = port?.speed ? port.speed : (speedOptions[0] || 'Standard');

        typeSelect.addEventListener('change', () => {
          const nextOptions = getPortSpeedOptions(typeSelect.value);
          speedSelect.innerHTML = '';
          nextOptions.forEach((speed) => {
            const option = document.createElement('option');
            option.value = speed;
            option.textContent = speed;
            speedSelect.appendChild(option);
          });
          speedSelect.value = nextOptions[0] || 'Standard';
        });

        const amountInput = document.createElement('input');
        amountInput.type = 'number';
        amountInput.min = '1';
        amountInput.max = '99';
        amountInput.value = port?.amount ? port.amount : 1;
        amountInput.dataset.role = 'amount';

        const removeBtn = document.createElement('button');
        removeBtn.type = 'button';
        removeBtn.textContent = '×';
        removeBtn.addEventListener('click', () => {
          if (ui.portConfig.children.length > 1) row.remove();
        });

        row.appendChild(typeSelect);
        row.appendChild(sideSelect);
        row.appendChild(speedSelect);
        row.appendChild(amountInput);
        row.appendChild(removeBtn);
        return row;
      }

      function populatePortConfig(device = null) {
        ui.portConfig.innerHTML = '';
        if (!device || !device.ports || device.ports.length === 0) {
          const preset = devicePortPresets[ui.deviceType.value] || devicePortPresets.Sonstiges;

        });

        grouped.forEach((port) => {
          const row = createPortRow({ ...port, amount: port.amount || 1 });
          ui.portConfig.appendChild(row);
        });
      }
